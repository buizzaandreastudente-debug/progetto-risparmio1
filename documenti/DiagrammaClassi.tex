  registrazione() : void \\ login() : bool \\ visualizzaStoricoPrezzi() : void
}

% I dati personali appartengono solo al cliente registrato
\umlclass[x=0, y=-5.5]{ClienteRegistrato}{
  nome : string \\ cognome : string \\ email : string \\ password : string
}{
  visualizzaOfferte() : void \\
  cercaOfferte(testo : string) : void \\
  ordinaPerPrezzo() : void \\
  filtraPerMarca(marca : string) : void
}

\umlclass[x=-1, y=-11]{GPS}{}{
  getPosizione() : Posizione
}

% --- Colonna centrale ---
\umlclass[x=6.5, y=-1]{Offerta}{
  descrizione : string \\ prezzo : float
}{}

\umlclass[x=6.5, y=-5.5]{StoricoPrezzo}{
  data : datetime \\ prezzo : float
}{}

\umlclass[x=6.5, y=-11]{Posizione}{
  latitudine : float \\ longitudine : float
}{}

% --- Colonna destra ---
\umlclass[x=12.5, y=-1]{Prodotto}{
  nome : string \\ descrizione : string
}{}

\umlclass[x=12.5, y=-5.5]{Categoria}{
  nome : string
}{}

\umlclass[x=18.5, y=-1]{Marca}{
  nome : string
}{}

\umlassoc[mult1=*, mult2=*]{ClienteRegistrato}{Offerta}

% il GPS fornisce le posizioni (non e' una relazione 1-1)
\umlassoc[name=fornisce, mult1=1, mult2=*]{GPS}{Posizione}

\umlassoc[mult1=*, mult2=1]{Offerta}{Prodotto}

\umlassoc[mult1=*, mult2=1]{Prodotto}{Categoria}

\umlassoc[mult1=*, mult2=1]{Prodotto}{Marca}
